Directory: sudo nano /etc/systemd/system/splu.service

[Unit]
Description=SPLU Python Service
After=network.target

[Service]
User=admin
WorkingDirectory=/home/admin/SPLU_PY
ExecStart=/home/admin/SPLU_PY/venv/bin/python main.py
Restart=always
RestartSec=5
Environment=PYTHONUNBUFFERED=1

Environment=DISPLAY=:0
Environment=XAUTHORITY=/home/admin/.Xauthority
Environment=XDG_RUNTIME_DIR=/run/user/1000
Environment=WAYLAND_DISPLAY=wayland-0
ExecStop=/usr/bin/pkill -f main.py
KillMode=process

[Install]
WantedBy=multi-user.target
